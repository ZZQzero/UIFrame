cmake_minimum_required(VERSION 3.22)
project(UIFrameSqlite VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Keep executable/DLL paths identical for single- and multi-configuration generators.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<0:>")
if(WIN32 AND (NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$"))
  message(FATAL_ERROR "The Windows plugin currently targets x86_64 only")
endif()
option(UFSQLITE_TESTS "Build native contract tests" ON)
option(UFSQLITE_SANITIZE "Enable address/undefined behavior sanitizer" OFF)
set(UFSQLITE_DEFINITIONS SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DQS=0
    SQLITE_DEFAULT_MEMSTATUS=1 SQLITE_USE_URI=0 SQLITE_ENABLE_LOCKING_STYLE=0)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/include/ufsqlite.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/source-lock.json")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp" UFSQLITE_CORE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/include/ufsqlite.h" UFSQLITE_ABI_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/source-lock.json" UFSQLITE_ENGINE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" UFSQLITE_CONFIG_HASH)
string(SHA256 UFSQLITE_BUILD_HASH "${UFSQLITE_CORE_HASH};${UFSQLITE_ABI_HASH};${UFSQLITE_ENGINE_HASH};${UFSQLITE_DEFINITIONS};${UFSQLITE_CONFIG_HASH}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/build-id.txt" CONTENT "${UFSQLITE_BUILD_HASH}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/compiler-info.txt" CONTENT
  "C: ${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}\nCXX: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\nSystem: ${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}\n")
function(ufsqlite_configure target)
  target_include_directories(${target} PUBLIC include PRIVATE third_party/sqlite)
  target_compile_definitions(${target} PRIVATE ${UFSQLITE_DEFINITIONS} UFSQLITE_BUILD_ID="${UFSQLITE_BUILD_HASH}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /FI${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/symbols.h
      $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  else()
    target_compile_options(${target} PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/symbols.h -Wall -Wextra)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Werror>)
  endif()
  if(WIN32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    if(MINGW)
      # Bundle the C++/unwind runtimes; only Windows system DLLs may be required.
      target_link_options(${target} PRIVATE -static)
    endif()
  endif()
  if(UFSQLITE_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(UFSQLITE_LINKAGE STATIC)
else()
  set(UFSQLITE_LINKAGE SHARED)
endif()
add_library(uiframe_sqlite ${UFSQLITE_LINKAGE} src/core.cpp third_party/sqlite/sqlite3.c)
ufsqlite_configure(uiframe_sqlite)
target_compile_definitions(uiframe_sqlite PRIVATE UFSQLITE_BUILD)
if(WIN32)
  set_target_properties(uiframe_sqlite PROPERTIES PREFIX "")
  if(MINGW)
    target_compile_options(uiframe_sqlite PRIVATE -gcodeview)
    target_link_options(uiframe_sqlite PRIVATE -Wl,--exclude-all-symbols
      "-Wl,--pdb=${CMAKE_BINARY_DIR}/uiframe_sqlite.pdb")
  endif()
  set_target_properties(uiframe_sqlite PROPERTIES PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<0:>")
endif()
if(ANDROID)
  target_link_options(uiframe_sqlite PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
endif()
if(UFSQLITE_TESTS)
  enable_testing()
  add_executable(ufsqlite_tests tests/core_tests.cpp)
  ufsqlite_configure(ufsqlite_tests)
  target_link_libraries(ufsqlite_tests PRIVATE uiframe_sqlite)
  add_test(NAME core_contract COMMAND ufsqlite_tests)
  add_executable(ufsqlite_io_failures tests/io_failures.cpp third_party/sqlite/sqlite3.c)
  ufsqlite_configure(ufsqlite_io_failures)
  target_compile_definitions(ufsqlite_io_failures PRIVATE UFSQLITE_BUILD)
  add_test(NAME io_failures COMMAND ufsqlite_io_failures)
  add_executable(ufsqlite_memory_failures tests/memory_failures.cpp third_party/sqlite/sqlite3.c)
  ufsqlite_configure(ufsqlite_memory_failures)
  target_compile_definitions(ufsqlite_memory_failures PRIVATE UFSQLITE_BUILD)
  add_test(NAME memory_failures COMMAND ufsqlite_memory_failures)
  set_tests_properties(core_contract io_failures memory_failures PROPERTIES TIMEOUT 120)
  if(WIN32)
    add_executable(ufsqlite_windows_lifecycle tests/windows_lifecycle.cpp)
    ufsqlite_configure(ufsqlite_windows_lifecycle)
    add_dependencies(ufsqlite_windows_lifecycle uiframe_sqlite)
    add_test(NAME windows_lifecycle COMMAND ufsqlite_windows_lifecycle)
    set_tests_properties(windows_lifecycle PROPERTIES TIMEOUT 30)
  endif()
endif()
