cmake_minimum_required(VERSION 3.22)
project(UIFrameBackup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(UFBACKUP_TESTS "Build shared repository contract tests" ON)
option(UFBACKUP_SANITIZE "Enable address/undefined behavior sanitizer" OFF)
set(UFSQLITE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Sqlite/Native~")
if(NOT UFSQLITE_LIBRARY)
  message(FATAL_ERROR "UFSQLITE_LIBRARY must name the already-built shared core artifact")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/catalog_schema.hpp"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build/schema.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/schema/catalog.sql" "${CMAKE_CURRENT_BINARY_DIR}/catalog_schema.hpp"
  DEPENDS schema/catalog.sql build/schema.py)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(UFBACKUP_LINKAGE STATIC)
else()
  set(UFBACKUP_LINKAGE SHARED)
endif()
add_library(uiframe_backup ${UFBACKUP_LINKAGE} src/repository.cpp "${CMAKE_CURRENT_BINARY_DIR}/catalog_schema.hpp")
target_include_directories(uiframe_backup PUBLIC include PRIVATE "${UFSQLITE_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(uiframe_backup PRIVATE UFBACKUP_BUILD)
target_link_libraries(uiframe_backup PRIVATE "${UFSQLITE_LIBRARY}")
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  # Package Editor path and built Player path both resolve the one engine.
  set_target_properties(uiframe_backup PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH "@loader_path;@loader_path/../../../Sqlite/Plugins/macOS")
endif()
if(WIN32)
  set_target_properties(uiframe_backup PROPERTIES PREFIX "")
  target_compile_definitions(uiframe_backup PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MINGW)
    target_link_options(uiframe_backup PRIVATE -static -Wl,--exclude-all-symbols)
  endif()
endif()
if(ANDROID)
  target_sources(uiframe_backup PRIVATE src/android.cpp)
  target_link_options(uiframe_backup PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
endif()
if(MSVC)
  target_compile_options(uiframe_backup PRIVATE /utf-8 /EHsc /W4)
else()
  target_compile_options(uiframe_backup PRIVATE -Wall -Wextra -Werror)
endif()
if(UFBACKUP_SANITIZE)
  target_compile_options(uiframe_backup PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(uiframe_backup PRIVATE -fsanitize=address,undefined)
endif()
if(UFBACKUP_TESTS)
  enable_testing()
  add_executable(ufbackup_tests tests/repository_tests.cpp)
  target_include_directories(ufbackup_tests PRIVATE "${UFSQLITE_ROOT}/include")
  target_link_libraries(ufbackup_tests PRIVATE uiframe_backup "${UFSQLITE_LIBRARY}")
  if(UFBACKUP_SANITIZE)
    target_compile_options(ufbackup_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ufbackup_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME backup_repository COMMAND ufbackup_tests)
  set_tests_properties(backup_repository PROPERTIES TIMEOUT 120)
endif()
